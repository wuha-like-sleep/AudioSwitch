cmake_minimum_required(VERSION 3.24)
project(AudioSwitchLoginProvider LANGUAGES CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(LoginProvider SHARED Provider.cpp Version.rc exports.def)
set_target_properties(LoginProvider PROPERTIES OUTPUT_NAME "AudioSwitch.LoginProvider")
target_compile_features(LoginProvider PRIVATE cxx_std_17)
target_compile_definitions(LoginProvider PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(LoginProvider PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/utf-8;/guard:cf;/sdl>")
target_link_options(LoginProvider PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
target_link_libraries(LoginProvider PRIVATE credui secur32 ole32 uuid shlwapi)
add_executable(LoginProviderTests Tests.cpp)
target_compile_features(LoginProviderTests PRIVATE cxx_std_17)
target_compile_definitions(LoginProviderTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(LoginProviderTests PRIVATE /W4 /WX /utf-8)
target_link_libraries(LoginProviderTests PRIVATE credui secur32 ole32 uuid shlwapi)
